# NLP pipeline runnable commands (from project root)
#
# Assumed working directory:
#   /home/aayush/Desktop/Projects/FKGIS
#
# These cover:
#   - Preprocessing of raw narrative / biography / interview text
#   - NER over processed JSON
#   - Dependency parsing over processed JSON
#   - Coreference global entity pool (legacy narrative)
#   - High-level pipeline for narrative (up to events)


# ============================================================
# 1) PREPROCESSING: RAW TEXT → processed_doc JSON
# ============================================================

# 1a) Narrative (Reporting Officer narrative)
# Input: raw narrative text, e.g. sampleRO.txt
# Output: sampleRO_ROnarrative_processed.json
preprocess narrative:
python3 FKGIS/nlp_pipeline/preprocessing/prep_ROnarrative.py FKGIS/nlp_pipeline/sampleRO.txt

# 1b) Biography
# Input: raw biography text, e.g. sampleBIO.txt
# Output: sampleBIO_bio_processed.json
preprocess biography:
python3 FKGIS/nlp_pipeline/preprocessing/prep_biography.py FKGIS/nlp_pipeline/sampleBIO.txt

# 1c) Interview (dialogue)
# Input: raw interview text, e.g. sampleInterview.txt
# Output: sampleInterview_interview_processed.json
preprocess interview:
python3 FKGIS/nlp_pipeline/preprocessing/prep_interview.py FKGIS/nlp_pipeline/sampleInterview.txt


# ============================================================
# 2) NER: processed_doc JSON → *_ner.json
# ============================================================

# Generic NER over any processed_doc JSON (narrative/bio/interview)
# Input: *_ROnarrative_processed.json, *_bio_processed.json, *_interview_processed.json
# Output: same name with "_ner.json" suffix
ner over processed_doc:
python3 FKGIS/nlp_pipeline/ner_and_dependency_parsing/ner_module.py --input PATH/TO/processed_doc.json

# Example (narrative):
# python3 FKGIS/nlp_pipeline/ner_and_dependency_parsing/ner_module.py --input FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed.json

# Example (biography):
# python3 FKGIS/nlp_pipeline/ner_and_dependency_parsing/ner_module.py --input FKGIS/nlp_pipeline/sampleBIO_bio_processed.json

# Example (interview):
# python3 FKGIS/nlp_pipeline/ner_and_dependency_parsing/ner_module.py --input FKGIS/nlp_pipeline/sampleInterview_interview_processed.json


# ============================================================
# 3) DEPENDENCY PARSING: attach "tokens" deps to sentences
# ============================================================

# Generic dependency parsing over any processed_doc JSON (with or without NER)
# Input: any processed_doc JSON
# Output: same name with "_dep.json" suffix, with sentence["tokens"] and ["root_index"]
dependency parsing:
python3 FKGIS/nlp_pipeline/dependency_parsing/dep_module.py --input PATH/TO/processed_doc_or_ner.json

# Example (narrative, after NER):
# python3 FKGIS/nlp_pipeline/dependency_parsing/dep_module.py --input FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed_ner.json


# ============================================================
# 4) COREFERENCE GLOBAL POOL (legacy single narrative doc)
# ============================================================

# Build a case-level global entity pool from a NER-enriched narrative JSON
# Input: *ROnarrative_processed_ner.json
# Output: {CASE_ID}_global_entities.json
coreference (global entities from narrative):
python3 FKGIS/nlp_pipeline/coreference/coref_global_pool.py --case CASE123 --ner_json FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed_ner.json


# ============================================================
# 5) HIGH-LEVEL PIPELINES
# ============================================================
#
# A) Narrative-only (up to events)
# B) Full case pipeline (narrative + biography + interviews)


# 5a) Only preprocessing via pipeline (narrative)
python3 -m FKGIS.nlp_pipeline.pipeline --input FKGIS/nlp_pipeline/sampleRO.txt --step preprocess
 
# 5b) Only NER via pipeline (assumes narrative preprocessed JSON already exists)
python3 -m FKGIS.nlp_pipeline.pipeline --input FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed.json --step ner
 
# 5c) Only coreference via pipeline (assumes narrative NER JSON already exists)
python3 -m FKGIS.nlp_pipeline.pipeline --case CASE123 --input FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed_ner.json --step coref
 
# 5d) Full narrative pipeline:
#     raw narrative text
#       → preprocessed JSON
#       → NER JSON
#       → case-level global entity pool (CASE123_global_entities.json)
#       → document-level coref_entities
#       → pronoun substitution + mention unification (unified_sentences)
#       → rule-based relations (relations)
#       → events with time_raw/time_resolved (events)
#
# Outputs:
#   - FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed.json
#   - FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed_ner.json
#   - CASE123_global_entities.json
#   - FKGIS/nlp_pipeline/sampleRO_ROnarrative_processed_ner_events.json
full narrative pipeline (up to events):
python3 -m FKGIS.nlp_pipeline.pipeline --case CASE123 --input FKGIS/nlp_pipeline/sampleRO.txt --step full


# 5e) Full multi-document case pipeline (narrative + bio + interviews):
#     1) load_docs from nlp_pipeline/case_docs/
#     2) preprocessing (type-specific)
#     3) NER
#     4) dependency parse
#     5) coreference
#     6) mention unification
#     7) relation extraction (forensic, rule-based)
#     8) event extraction (participants, locations, event_ids)
#     9) timeline builder (time_resolved + BEFORE edges)
#    10) KG builder (graph_nodes.json, graph_edges.json)
#    11) verification (sanity checks)
#
# Outputs (under nlp_pipeline/output/):
#   - case_output.json            # summary of documents, timeline, KG and verification
#   - graph_nodes.json            # nodes ready for Neo4j ingestion
#   - graph_edges.json            # edges ready for Neo4j ingestion
#   - *_enriched.json per doc    # enriched narrative/bio/interview docs
full case pipeline:
python3 -m FKGIS.nlp_pipeline.pipeline --case CASE123 --step case
